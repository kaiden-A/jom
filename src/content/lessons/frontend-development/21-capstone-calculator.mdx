This is it — the whole vanilla JavaScript app.

We have the HTML, the CSS, and the math. Now we connect them:

1. **State** — the target, the numbers, and the food list.
2. **Render** — draw the page from state.
3. **Events** — the forms and the remove buttons.
4. **Storage** — save and load the state.

The pattern is exactly what we practiced in lessons 17 and 18. The only difference is that it's all in one app.

## The shape of the state

```js
let state = {
  target: 0,
  bmr: 0,
  tdee: 0,
  activity: "moderate",
  foods: [],
};
```

One object. Everything the UI needs is in there. When something changes, we update the state, save it, and call `render()`.

<Callout variant="tip" title="Derive, don't store">
The total calories and the remaining calories are **not** in the state. They're calculated from `state.foods` and `state.target` inside `render()`. If you store them too, you have three values that can drift apart.
</Callout>

## render() is the heart

```js
function render() {
  const eaten = totalCalories(state.foods);
  const left = state.target ? state.target - eaten : null;

  targetEl.textContent = state.target ? state.target.toLocaleString() : "0";
  eatenEl.textContent = eaten.toLocaleString();
  leftEl.textContent = left === null ? "—" : left.toLocaleString();

  list.replaceChildren();
  for (const food of state.foods) {
    // create a row, set textContent, append
  }
}
```

Every event handler ends with:

```js
saveState();
render();
```

Change the state, save it, redraw. That's the loop.

const calculatorForm = document.querySelector("#calculator-form");

const foodName = document.querySelector("#food-name");
const foodCalories = document.querySelector("#food-calories");
const list = document.querySelector("#food-list");

const targetEl = document.querySelector("#target");
const bmrEl = document.querySelector("#bmr");
const tdeeEl = document.querySelector("#tdee");
const eatenEl = document.querySelector("#eaten");
const leftEl = document.querySelector("#left");

const countEl = document.querySelector("#count");
const totalEl = document.querySelector("#total");
const emptyEl = document.querySelector("#empty");
const clearButton = document.querySelector("#clear-log");

let state = loadState();

function activityMultiplier(level) {
  if (level === "sedentary") return 1.2;
  if (level === "light") return 1.375;
  if (level === "moderate") return 1.55;
  if (level === "active") return 1.725;
  return 1.2;
}

function calculateBMR(weight, height, age, sex) {
  const base = 10 * weight + 6.25 * height - 5 * age;
  return sex === "male" ? base + 5 : base - 161;
}

function calculateTarget({ weight, height, age, sex, activity, goal }) {
  const bmr = calculateBMR(weight, height, age, sex);
  const tdee = bmr * activityMultiplier(activity);

  let adjustment = 0;
  if (goal === "lose") adjustment = -500;
  if (goal === "gain") adjustment = 300;

  return {
    bmr: Math.round(bmr),
    tdee: Math.round(tdee),
    target: Math.round(tdee + adjustment),
  };
}

function totalCalories(foods) {
  return foods.reduce((sum, food) => sum + food.calories, 0);
}

function render() {
  const eaten = totalCalories(state.foods);
  const left = state.target ? state.target - eaten : null;

  targetEl.textContent = state.target ? state.target.toLocaleString() : "0";
  bmrEl.textContent = state.bmr ? state.bmr.toLocaleString() : "—";
  tdeeEl.textContent = state.tdee ? state.tdee.toLocaleString() : "—";
  eatenEl.textContent = eaten.toLocaleString();
  leftEl.textContent = left === null ? "—" : left.toLocaleString();
  leftEl.classList.toggle("over", left !== null && left < 0);

  summaryEl.textContent = state.target
    ? "Based on your details and " + state.activity + " activity."
    : "Fill in your details and press Calculate.";

  list.replaceChildren();

  for (const food of state.foods) {
    const row = document.createElement("tr");

    const nameCell = document.createElement("td");
    nameCell.textContent = food.name;

    const calorieCell = document.createElement("td");
    calorieCell.textContent = food.calories.toLocaleString();

    const actionCell = document.createElement("td");
    const remove = document.createElement("button");
    remove.type = "button";
    remove.className = "remove";
    remove.textContent = "Remove";
    remove.dataset.id = food.id;
    remove.setAttribute("aria-label", "Remove " + food.name);
    actionCell.append(remove);

    row.append(nameCell, calorieCell, actionCell);
    list.append(row);
  }

  totalEl.textContent = eaten.toLocaleString();
  countEl.textContent = state.foods.length + (state.foods.length === 1 ? " item" : " items");
  emptyEl.hidden = state.foods.length > 0;
}

function saveState() {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
}

function loadState() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) {
      return { target: 0, bmr: 0, tdee: 0, activity: "moderate", foods: [] };
    }

    const parsed = JSON.parse(raw);
    return {
      target: Number(parsed.target) || 0,
      bmr: Number(parsed.bmr) || 0,
      tdee: Number(parsed.tdee) || 0,
      activity: parsed.activity || "moderate",
      foods: Array.isArray(parsed.foods) ? parsed.foods : [],

    return { target: 0, bmr: 0, tdee: 0, activity: "moderate", foods: [] };

}

  const height = Number(document.querySelector("#height").value);

## Read the code in this order

- **State** — `let state = loadState()`. One object, loaded from storage.
- **Helpers** — `activityMultiplier`, `calculateBMR`, `calculateTarget`, `totalCalories`. Pure functions.
- **render()** — reads state, writes the DOM. Total and left are derived, not stored.
- **Events** — calculator submit, food submit, remove (delegated), clear.
- **Startup** — `render()` once, then wait for events.

Every handler follows the same rhythm:

```js
// 1. prevent the default
// 2. read and validate input
// 3. update state
// 4. save
// 5. render
```

Once you notice that pattern, the app stops looking like a lot of code.

<Callout variant="warn" title="No innerHTML for user data">
Food names come from the user. We create a `<td>` and set `textContent`, rather than building an HTML string. If someone types `<img src=x onerror=alert(1)>`, it shows up as text — not code. Keep doing this.
</Callout>

## The mistakes everyone makes

- **Forgetting to call render.** The state changes; the screen doesn't.
- **Forgetting to save after delete.** The page looks right until you refresh.
- **Using innerHTML for food names.** XSS. Use textContent.
- **Storing the total separately.** Derive it from foods.
- **Not validating numbers.** `Number("abc")` is `NaN`; a `NaN` total poisons everything.
- **Dividing by a zero target.** Check for 0 before calculating a percentage.
- **One giant event handler.** Keep calculate, add, remove, and clear separate.
- **Assuming localStorage is a database.** It's per browser and easy to clear.
